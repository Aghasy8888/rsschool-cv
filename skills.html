Valid Parentheses — check if brackets are correctly matched using a stack.

const isValid = function(s) {
  const stack = [];
  const bracketMap = {
    ')': '(',
    '}': '{',
    ']': '[',
  };

  for (let char of s) {
    if (bracketMap[char]) {
      if (stack.pop() !== bracketMap[char]) {
        return false;
      }
    } else {
      stack.push(char);
    }
  }

  return stack.length === 0;
};
